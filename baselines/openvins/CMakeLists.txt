# ov_run (OpenVINS on a sequence folder) against an ov_msckf_lib built without ROS (build_openvins.sh)
cmake_minimum_required(VERSION 3.10)
project(ov_run)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time)
find_package(Ceres REQUIRED)
add_definitions(-DROS_AVAILABLE=0 -DENABLE_ARUCO_TAGS=0)
add_executable(ov_run ov_run.cpp)
target_include_directories(ov_run PRIVATE ${OPENVINS_SRC}/ov_core/src ${OPENVINS_SRC}/ov_init/src ${OPENVINS_SRC}/ov_msckf/src
                           ${EIGEN3_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS})
target_link_libraries(ov_run ${OPENVINS_LIB} ${OpenCV_LIBRARIES} ${Boost_LIBRARIES} Ceres::ceres)
